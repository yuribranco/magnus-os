import type { ScoreInput, Scored } from "./types.js";

/** Normaliza 0..1, tratando NaN/inf/negativo como 0 e clampando teto em 1. */
function frac(value: number, max: number): number {
  if (!Number.isFinite(value) || value <= 0) return 0;
  return Math.min(value / max, 1);
}

const LONGEVITY_MAX_DAYS = 90; // ad 30d+ ≈ lucrativo; 90d = teto
const VARIATION_MAX = 10;      // 10+ variações reais = teto
const VELOCITY_MAX = 5;        // 5+ lançamentos em 14d = teto

const W_LONGEVITY = 0.5;
const W_VARIATION = 0.3;
const W_VELOCITY = 0.2;

export function persistenceScore(input: ScoreInput): Scored {
  const longevity = frac(input.daysActive, LONGEVITY_MAX_DAYS);
  const variation = frac(input.variations, VARIATION_MAX);
  const velocity = frac(input.recentLaunches14d, VELOCITY_MAX);
  const score = W_LONGEVITY * longevity + W_VARIATION * variation + W_VELOCITY * velocity;
  return { score, parts: { longevity, variation, velocity } };
}
```

import { createHash } from "node:crypto";

export interface RawAd {
  ad_archive_id: string;
  page_id: string;
  page_name: string;
  started_at: string;        // YYYY-MM-DD
  is_active: boolean;
  variations: number;
  media_type: string;
  cta_text: string;
  hook: string;
  link_url: string;
  thumb_url: string;
  creative_fp: string;
}

const s = (v: unknown): string => (v === null || v === undefined ? "" : String(v));
const n = (v: unknown): number => { const x = Number(v); return Number.isFinite(x) ? x : 0; };

export function advertiserUrl(pageId: string): string {
  const id = encodeURIComponent(s(pageId));
  return `https://www.facebook.com/ads/library/?active_status=active&ad_type=all&country=BR&view_all_page_id=${id}`;
}

export function keywordUrl(keyword: string): string {
  const q = encodeURIComponent(s(keyword));
  return `https://www.facebook.com/ads/library/?active_status=active&ad_type=all&country=BR&q=${q}&search_type=keyword_unordered&media_type=all`;
}

export function parseAd(raw: any): RawAd | null {
  const archive = s(raw?.adArchiveId ?? raw?.adArchiveID);
  if (!archive) return null;
  const snap = raw?.snapshot ?? {};
  const started = s(raw?.startDateFormatted).slice(0, 10); // YYYY-MM-DD
  const thumb = s(snap?.videos?.[0]?.videoPreviewImageUrl || snap?.images?.[0]?.originalImageUrl || "");
  const hook = s(snap?.body?.text).split("\n")[0]?.trim() ?? "";
  const fp = createHash("sha256")
    .update([s(snap?.body?.text), s(snap?.linkUrl), s(snap?.displayFormat)].join("|"))
    .digest("hex").slice(0, 16);
  return {
    ad_archive_id: archive,
    page_id: s(raw?.pageId ?? raw?.pageID),
    page_name: s(raw?.pageName),
    started_at: started,
    is_active: Boolean(raw?.isActive),
    variations: n(raw?.collationCount) || 1,
    media_type: s(snap?.displayFormat) || "UNKNOWN",
    cta_text: s(snap?.ctaText),
    hook,
    link_url: s(snap?.linkUrl),
    thumb_url: thumb,
    creative_fp: fp,
  };
}
```

import { createClient } from "@supabase/supabase-js";
import type { RawAd } from "./apify.js";

const TTL_MS = 7 * 24 * 60 * 60 * 1000;

export function isFresh(lastScrapedAt: string | null, nowMs: number = Date.now()): boolean {
  if (!lastScrapedAt) return false;
  return nowMs - new Date(lastScrapedAt).getTime() < TTL_MS;
}

/** ids presentes no DB mas ausentes do scrape novo → devem virar inativos. */
export function diffActiveIds(dbIds: string[], scrapedIds: string[]): string[] {
  const seen = new Set(scrapedIds);
  return dbIds.filter((id) => !seen.has(id));
}

function client() {
  const url = process.env.SUPABASE_URL, key = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !key) throw new Error("SUPABASE_URL/SUPABASE_SERVICE_ROLE_KEY ausentes");
  return createClient(url, key, { db: { schema: "radar" } });
}

/** Upsert do anunciante + ads; marca lifecycle. Throw com contexto (Supabase não throw sozinho). */
export async function upsertAdvertiserAds(pageId: string, pageName: string, source: string, ads: RawAd[]) {
  const db = client();
  const { data: adv, error: e1 } = await db.from("advertisers")

  if (e1) throw new Error(`upsert advertiser ${pageId}: ${e1.message}`);

  const nowIso = new Date().toISOString();
  const rows = ads.map((a) => ({
    advertiser_id: adv!.id, ad_archive_id: a.ad_archive_id, body_text: a.hook, cta_text: a.cta_text,
    link_url: a.link_url, thumb_url: a.thumb_url, media_type: a.media_type, started_at: a.started_at,
    creative_fp: a.creative_fp, collation_count: a.variations, is_active: true,
    last_verified_active_at: nowIso, last_seen_at: nowIso, inactive_at: null,
  }));
  if (rows.length) {
    const { error: e2 } = await db.from("ads").upsert(rows, { onConflict: "ad_archive_id" });
    if (e2) throw new Error(`upsert ads ${pageId}: ${e2.message}`);
  }
  // lifecycle: ads do anunciante ausentes do scrape novo → inativos
  const { data: existing } = await db.from("ads").select("ad_archive_id").eq("advertiser_id", adv!.id).eq("is_active", true);
  const stale = diffActiveIds((existing ?? []).map((r) => r.ad_archive_id), ads.map((a) => a.ad_archive_id));
  if (stale.length) {
    const { error: e3 } = await db.from("ads").update({ is_active: false, inactive_at: nowIso }).in("ad_archive_id", stale);
    if (e3) throw new Error(`lifecycle ${pageId}: ${e3.message}`);
  }
  return adv!.id as string;
}

export function buildRadarJson(niche: string, ads: EnrichedAd[], nowMs: number = Date.now()) {

    const daysActive = Math.max(0, Math.round((nowMs - new Date(a.started_at).getTime()) / DAY));

      score, score_parts: parts, label: null as string | null,
    };
  }).sort((x, y) => y.score - x.score);

const DAY = 24 * 60 * 60 * 1000;

function enrich(ads: RawAd[], nowMs: number): EnrichedAd[] {
  const recent = ads.filter((a) => nowMs - new Date(a.started_at).getTime() < 14 * DAY).length;

export function distinctAdvertisers(ads: { page_id: string; page_name: string }[], cap: number) {
  const seen = new Set<string>(); const out: { page_id: string; page_name: string }[] = [];

import fs from "node:fs";
import { radarFilePath } from "./paths";
import type { RadarData, RadarAd } from "./radar-types";

function readJson<T>(fp: string): T | null {
  try { return JSON.parse(fs.readFileSync(fp, "utf8")) as T; } catch { return null; }
}
const str = (v: unknown) => (v == null ? "" : String(v));
const num = (v: unknown) => { const x = Number(v); return Number.isFinite(x) ? x : 0; };

export function parseRadar(raw: unknown): RadarData | null {
  if (!raw || typeof raw !== "object") return null;
  const r = raw as any;
  if (!Array.isArray(r.ads)) return null;
  const ads: RadarAd[] = r.ads
    .filter((a: any) => a && a.ad_archive_id)
    .map((a: any) => ({
      ad_archive_id: str(a.ad_archive_id),
      advertiser: { page_id: str(a.advertiser?.page_id), page_name: str(a.advertiser?.page_name), source: str(a.advertiser?.source) },
      started_at: str(a.started_at), days_active: num(a.days_active), is_active: Boolean(a.is_active ?? true),
      variations: num(a.variations), media_type: str(a.media_type), cta_text: str(a.cta_text),
      hook: str(a.hook), link_url: str(a.link_url), thumb_url: str(a.thumb_url),
      score: num(a.score),
      score_parts: { longevity: num(a.score_parts?.longevity), variation: num(a.score_parts?.variation), velocity: num(a.score_parts?.velocity) },
      label: a.label === "util" || a.label === "inutil" ? a.label : null,
    }))
    .sort((x: RadarAd, y: RadarAd) => y.score - x.score);
  return { generated_at: str(r.generated_at), niche: str(r.niche), ads };
}

export function readRadar(slug: string): RadarData | null {
  return parseRadar(readJson<unknown>(radarFilePath(slug)));
}
