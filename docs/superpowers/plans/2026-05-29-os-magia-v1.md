import fs from "node:fs";
import path from "node:path";

export function isImageRel(rel: string): boolean {
  return /\.(png|jpe?g|webp|gif|svg|avif)$/i.test(rel);
}

/** relPaths (posix) de arquivos em operacao/<slug>/ com mtime >= sinceMs. Vazio se nada. */
export function listOutputsSince(root: string, slug: string, sinceMs: number): string[] {
  const base = path.join(root, "operacao", slug);
  if (!slug || !fs.existsSync(base)) return [];
  const out: string[] = [];
  const walk = (dir: string) => {
    for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
      const abs = path.join(dir, e.name);
      if (e.isDirectory()) { walk(abs); continue; }
      try {
        if (fs.statSync(abs).mtimeMs >= sinceMs) {
          out.push(path.relative(root, abs).split(path.sep).join("/"));
        }
      } catch { /* arquivo sumiu no meio do walk */ }
    }
  };
  walk(base);
  // imagens primeiro (são o "wow"), depois o resto, estável
  return out.sort((a, b) => Number(isImageRel(b)) - Number(isImageRel(a)));
}
```

import { isImageRel } from "@/lib/run-outputs";
export function DoneSummary({ path: summaryPath, outputs, onOpenFolder, onRerun, title }: {
  path?: string; outputs: string[]; onOpenFolder?: () => void; onRerun?: () => void; title?: string;
}) {
  const imgs = outputs.filter(isImageRel);
  const files = outputs.filter((o) => !isImageRel(o));
  const href = (rel: string) => `/api/files?p=${encodeURIComponent(rel)}`;
  return (
    <div className="done-summary">
      <div className="done-summary__title"><Icons.check size={14} /> {title ?? "Pronto."}</div>
      {imgs.length > 0 && (
        <div className="done-summary__imgs">
          {imgs.map((o) => (
            <a key={o} href={href(o)} target="_blank" rel="noreferrer" className="done-summary__thumb"
               style={{ backgroundImage: `url("${href(o)}")` }} title={o.split("/").pop()} />
          ))}
        </div>
      )}
      {files.map((o) => (
        <a key={o} className="done-summary__link" href={href(o)} target="_blank" rel="noreferrer">
          <Icons.file size={11} /> {o.split("/").pop()}
        </a>
      ))}
      {outputs.length === 0 && <div className="dim" style={{ fontSize: "var(--ts-sm)" }}>Concluído.</div>}
      <div className="permission__actions">
        {onOpenFolder && <button className="btn btn--secondary btn--sm" type="button" onClick={onOpenFolder}><Icons.folder size={12} /> Abrir pasta</button>}
        {onRerun && <button className="btn btn--ghost btn--sm" type="button" onClick={onRerun}><Icons.refresh size={12} /> Gerar variação</button>}
      </div>
    </div>
  );
}
```
